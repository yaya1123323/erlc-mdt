  <script>
    function showForm(id) {
      document.querySelectorAll('.form-section').forEach(section => {
        section.classList.remove('active');
      });
      document.getElementById(id).classList.add('active');
    }

    function randomCase(prefix) {
      return prefix + '-' + Math.floor(10000 + Math.random() * 90000);
    }

    function generateArrest() {
      document.getElementById('arrestOutput').innerText =
`CASE ID: ${randomCase('AR')}
Officer: ${aOfficer.value}
Badge: ${aBadge.value}
Suspect: ${aSuspect.value}
Charges: ${aCharges.value}
Location: ${aLocation.value}
Fine: ${aFine.value}
Jail Time: ${aJail.value}
Vehicle: ${aVehicle.value}
Narrative: ${aNarrative.value}`;
    }

    function generateTicket() {
      document.getElementById('ticketOutput').innerText =
`TICKET ID: ${randomCase('CT')}
Officer: ${tOfficer.value}
Driver: ${tDriver.value}
Violation: ${tViolation.value}
Vehicle: ${tVehicle.value}
Fine: ${tFine.value}
Location: ${tLocation.value}
Notes: ${tNotes.value}`;
    }

    function generateWarning() {
      document.getElementById('warningOutput').innerText =
`WARNING ID: ${randomCase('WR')}
Officer: ${wOfficer.value}
Citizen: ${wCitizen.value}
Reason: ${wReason.value}
Location: ${wLocation.value}
Type: ${wType.value}
Notes: ${wNotes.value}`;
    }

    function generateScene() {
      document.getElementById('sceneOutput').innerText =
`SCENE ID: ${randomCase('SC')}
Scene Type: ${sType.value}
Primary Officer: ${sOfficer.value}
Location: ${sLocation.value}
Units Involved: ${sUnits.value}
Arrests Made: ${sArrests.value}
Vehicles Towed: ${sTow.value}
Outcome: ${sOutcome.value}`;
    }
  function submitSupervisor(type, content) {
      if (!content || content.trim() === '') {
        alert('Generate a report first before submitting.');
        return;
      }

      const webhookMessage = `SUPERVISORY BOARD SUBMISSION

Type: ${type}

${content}`;

      navigator.clipboard.writeText(webhookMessage);

      alert('Report copied. Paste it into your Supervisory Review Discord channel or webhook system.');
    }
  </script>
